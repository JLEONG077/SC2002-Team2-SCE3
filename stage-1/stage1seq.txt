```plantuml
@startuml

title Restaurant Rush Stage 3 sequence diagram

hide footbox

skinparam shadowing false
skinparam sequenceMessageAlign left
skinparam responseMessageBelowArrow true

actor ":Manager" as M
boundary ":ConsoleUI" as UI
control ":GameController" as GC
control ":TurnProcessor" as TP
entity ":Restaurant" as R
entity ":HappyHourPricing" as P
entity ":Host" as H
entity ":Waiter" as W
entity ":VIPCustomer" as V
entity ":CriticCustomer" as X
entity ":Order" as O
control ":Cashier" as K
control ":ResultEvaluator" as RE

loop turn = 1..18 (turns 7 and 8 shown)

    activate GC
    GC -> TP : runTurn(turn)
    activate TP

    TP -> UI : showTurnStart(restaurant, turn)
    activate UI
    deactivate UI

    note over UI, O
      Turn 7: a VIP waits in the queue and one table is free.
      The manager may use Happy Hour once per game.
    end note

    opt &#91;Happy Hour not yet used&#93;
        M -> UI : chooses Happy Hour
        activate UI
        deactivate UI

        UI -> GC : activateHappyHour()
        activate GC

        GC -> P : new HappyHourPricing(2 turns)
        activate P
        deactivate P

        GC -> R : setPricing(policy)
        activate R

        R -> P : onTurnStart(restaurant)
        activate P

        P -> V : recoverSatisfaction(15)
        activate V
        deactivate V

        P --> R
        deactivate P

        R --> GC
        deactivate R

        GC --> UI : ActionResult(success)
        deactivate GC
    end

    note over UI, O
      Step 2: the Host and the Waiter each act this turn.
      The Host seats the VIP; the Waiter serves a READY dish.
    end note

    TP -> UI : promptHostAction(restaurant)
    activate UI

    M -> UI : chooses SEAT VIP
    activate UI
    deactivate UI

    UI --> TP : StaffAction(SEAT, vip)
    deactivate UI

    TP -> H : perform(action, restaurant, turn)
    activate H

    H -> R : seat(vip, freeTable)
    activate R

    R -> V : setState(SEATED)
    activate V
    deactivate V

    R --> H
    deactivate R

    H --> TP : ActionResult(success)
    deactivate H

    TP -> UI : promptWaiterAction(restaurant)
    activate UI

    M -> UI : chooses SERVE order
    activate UI
    deactivate UI

    UI --> TP : StaffAction(SERVE, order)
    deactivate UI

    TP -> W : perform(action, restaurant, turn)
    activate W

    W -> O : isServable(turn)
    activate O
    O --> W : true
    deactivate O

    W -> O : markServed()
    activate O
    deactivate O

    W --> TP : ActionResult(success)
    deactivate W

    note over UI, O
      Turn 8: the Waiter takes the VIP's order while
      Happy Hour is still active.
      Price: $15.00, then VIP 10% off, then Happy Hour 20% off.
    end note

    TP -> W : perform(TAKE_ORDER, restaurant, turn)
    activate W

    W -> V : chooseItem(menu)
    activate V
    V --> W : Burger
    deactivate V

    W -> V : discount(1500)
    activate V
    V --> W : 1350
    deactivate V

    W -> R : priceFor(1350)
    activate R

    R -> P : finalPrice(1350)
    activate P
    P --> R : 1080
    deactivate P

    R --> W : 1080
    deactivate R

    W -> O : new Order(Burger, 1080)
    activate O
    deactivate O

    W -> UI : showChoiceMade(vip, Burger)
    activate UI
    deactivate UI

    W --> TP : ActionResult(success)
    deactivate W

    TP --> GC : turn done
    deactivate TP

end

note over GC, RE
  After turn 18: result check
end note

GC -> RE : evaluate(restaurant)
activate RE

RE -> R : getRevenue(), averageSatisfaction()
activate R
R --> RE : values
deactivate R

alt &#91;paid >= 4 and revenue >= $45.00 and average >= 60&#93;
    RE --> GC : VICTORY
else &#91;otherwise&#93;
    RE --> GC : DEFEAT
end

deactivate RE

GC -> UI : showResult(outcome, restaurant)
activate UI
deactivate UI

GC --> UI
deactivate GC

note over UI, O
  Additional scenario: the Critic is seated and orders.
  chooseItem() is overridden in CriticCustomer.
end note

UI --> TP : StaffAction(SEAT, critic)
activate TP

TP -> W : perform(action, restaurant, turn)
activate W

W -> R : seat(critic, freeTable)
activate R

R -> X : setState(SEATED)
activate X
deactivate X

R --> W
deactivate R

W --> TP : ActionResult(success)
deactivate W

note over UI, O
  The Waiter takes the Critic's order.
end note

TP -> W : perform(TAKE_ORDER, restaurant, turn)
activate W

W -> X : chooseItem(menu)
activate X

X -> X : Critic rule (decisions log)
X --> W : Pasta
deactivate X

W -> X : discount(1800)
activate X
X --> W : 1800 (no discount)
deactivate X

W -> O : new Order(Pasta, 1800)
activate O
deactivate O

W -> UI : showChoiceMade(critic, Pasta)
activate UI
deactivate UI

W --> TP : ActionResult(success)
deactivate W

note over UI, O
  Step 5: waiting loss.
  The same call is made on every customer type, with no type checks.
end note

loop &#91;each waiting or seated unserved customer&#93;
    TP -> V : loseSatisfaction()
    activate V
    V -> V : satisfaction -= lossPerTurn() (5)
    V --> TP
    deactivate V

    TP -> X : loseSatisfaction()
    activate X
    X -> X : satisfaction -= lossPerTurn() (12)

    opt &#91;satisfaction reaches 0&#93;
        X -> O : cancel()
        activate O
        deactivate O

        X -> R : removeCustomer(this)
        activate R
        deactivate R
    end

    X --> TP
    deactivate X
end

note over UI, O
  Later turn: the VIP is seated and orders a Burger.
end note

TP -> W : perform(TAKE_ORDER, restaurant, turn)
activate W

W -> V : discount(1500)
activate V
V -> V : 90% of price
V --> W : 1350
deactivate V

W -> O : new Order(Burger, 1350)
activate O
deactivate O

W --> TP : ActionResult(success)
deactivate W

note over UI, O
  Payment: the Cashier never checks the customer type.
end note

TP -> K : collectPayment(restaurant, turn)
activate K

alt &#91;oldest READY_TO_PAY became ready on an earlier turn&#93;
    K -> O : markPaid()
    activate O
    deactivate O

    K -> R : recordPayment(vip)
    activate R

    R -> R : revenue += 1350, table.release()
    R --> K
    deactivate R

    K --> TP : ActionResult(success)
else &#91;none eligible&#93;
    K --> TP : nothing to collect
end

deactivate K
deactivate TP

@enduml
```
